# nginx 服务监控与恢复脚本

在目标服务器上以 root 执行以下内容，即可生成脚本文件 `/opt/nginx_service_monitor.sh`。

```bash
cat >/opt/nginx_service_monitor.sh<<'EOA'
#!/bin/bash

SERVICE="nginx"
PORT=80
LOG_FILE="/tmp/nginx_service_monitor.log"

while true; do
    # 进程存活检测
    if ! systemctl is-active --quiet "$SERVICE"; then
        echo "[$(date)] $SERVICE 进程不存在,尝试重启" >> "$LOG_FILE"
        systemctl restart "$SERVICE"
        sleep 5
        continue
    fi

    # 端口检测
    if ss -tln | grep -q ":$PORT "; then
        echo "[$(date)] $SERVICE 端口 $PORT 正常监听" >> "$LOG_FILE"
    else
        echo "[$(date)] $SERVICE 端口 $PORT 未监听,尝试重启" >> "$LOG_FILE"
        systemctl restart "$SERVICE"

        sleep 5

        if ss -tln | grep -q ":$PORT "; then
            echo "[$(date)] 恢复成功" >> "$LOG_FILE"
        else
            echo "[$(date)] 恢复失败" >> "$LOG_FILE"
        fi
    fi

    sleep 30
done
EOA
```

生成后运行：

```bash
bash /opt/nginx_service_monitor.sh
```
